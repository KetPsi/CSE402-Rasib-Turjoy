\subsection{Orthogonality}

\begin{definition}[Orthogonality]
	Let \( V \) be an inner product space. Two vectors \( \mathbf{x}, \mathbf{y} \in V \) are \emph{orthogonal} if
	\begin{equation}
		\langle \mathbf{x}, \mathbf{y} \rangle = 0.
		\label{eq:orthogonality-1}
	\end{equation}
\end{definition}



\begin{example}
	Let \( \mathbf{x} = \begin{bmatrix} 1 \\ 2 \\ 1 \end{bmatrix}, \mathbf{y} = \begin{bmatrix} 1 \\ -1 \\ 1 \end{bmatrix} \in \mathbb{R}^3 \). Then

	\begin{equation}
		\langle \mathbf{x}, \mathbf{y} \rangle = 1 \times 1 + 2 \times (-1) + 1 \times 1 = 0.
		\label{eq:orthogonality-2}
	\end{equation}

	Therefore, \( \mathbf{x}, \mathbf{y} \) are orthogonal/perpendicular.

	The two vectors are shown below.

\end{example}

But since inner product is defined for any \( \mathbb{R}^n \), we can talk about orthogonality there as well.

\begin{example}
	Let \( \mathbf{x} = \begin{bmatrix} 1 \\ 2 \\ 1 \\ 0 \\ -7 \end{bmatrix}, \mathbf{y} = \begin{bmatrix} 1 \\ -1 \\ 1 \\ 5 \\ 0 \end{bmatrix} \in \mathbb{R}^3 \). Then

	\begin{equation}
		\langle \mathbf{x}, \mathbf{y} \rangle = 1 \times 1 + 2 \times (-1) + 1 \times 1 + 0 \times 5 + (-7) \times 0 = 0.
		\label{eq:orthogonality-2.1}
	\end{equation}

	Therefore, \( \mathbf{x}, \mathbf{y} \) are orthogonal.

	Notice, \( \mathbf{x}, \mathbf{y} \) are vectors in \( 5 \)-dimensional space  \( \mathbb{R}^5 \), which we can not visualise, yet we can talk about whether they are orthogonal/perpendicular to each other using the language of (linear) algebra.
\end{example}

If the basis vectors $v_1,v_2,\ldots,v_n$ of the vector
space $\mathbb{R}^n$ are pairwise orthogonal, i.e.,
\begin{equation}
	\langle v_i,v_j\rangle=0,
	\qquad i\neq j,
	\label{eq:orthogonality-3}
\end{equation}
then the coefficients of any vector $v\in\mathbb{R}^n$
with respect to this basis can be determined easily.
Write
\begin{equation}
	v=\sum_{i=1}^{n}c_i v_i,
\end{equation}
and taking the inner product with $v_j$ on both sides,
we obtain
\begin{align}
	\langle v,v_j\rangle
	 & =\left\langle\sum_{i=1}^{n}c_i v_i,v_j\right\rangle \\
	 & =\sum_{i=1}^{n}c_i\langle v_i,v_j\rangle            \\
	 & =c_j\langle v_j,v_j\rangle,
\end{align}
since $\langle v_i,v_j\rangle=0$ whenever $i\neq j$.
Therefore,
\begin{equation}
	c_j=\frac{\langle v,v_j\rangle}
	{\langle v_j,v_j\rangle}
	=\frac{\langle v,v_j\rangle}{\|v_j\|^2}
	\label{eq:orthogonality-4}
\end{equation}
Thus, orthogonality allows us to determine each
coefficient independently, without solving a system
of linear equations. In particular, if the basis
is orthonormal, then
\begin{equation}
	v_j=\langle v,v_j\rangle
	\label{eq:orthogonality-5}
\end{equation}

\begin{example}
	Consider the following orthogonal basis of
	\(\mathbb{R}^3\):
	\[
		v_1=\begin{bmatrix}1\\1\\0\end{bmatrix},
		\qquad
		v_2=\begin{bmatrix}1\\-1\\0\end{bmatrix},
		\qquad
		v_3=\begin{bmatrix}0\\0\\2\end{bmatrix}.
	\]
	These vectors are pairwise orthogonal, with
	\[
		\|v_1\|^2=2,\qquad
		\|v_2\|^2=2,\qquad
		\|v_3\|^2=4.
	\]

	We want to express
	\(v=\begin{bmatrix}3\\1\\4\end{bmatrix}\)
	as \(v=c_1v_1+c_2v_2+c_3v_3\).
	Using \eqref{eq:orthogonality-4},
	\begin{align*}
		c_1
		 & =\frac{\langle v,v_1\rangle}{\|v_1\|^2}
		=\frac{3+1}{2}=2,                          \\
		c_2
		 & =\frac{\langle v,v_2\rangle}{\|v_2\|^2}
		=\frac{3-1}{2}=1,                          \\
		c_3
		 & =\frac{\langle v,v_3\rangle}{\|v_3\|^2}
		=\frac{4\cdot2}{4}=2.
	\end{align*}
	Therefore,
	\[
		v=2v_1+v_2+2v_3
		=\begin{bmatrix}2\\2\\0\end{bmatrix}
		+\begin{bmatrix}1\\-1\\0\end{bmatrix}
		+\begin{bmatrix}0\\0\\4\end{bmatrix}
		=\begin{bmatrix}3\\1\\4\end{bmatrix}.
	\]
	Each coefficient is obtained independently by
	taking an inner product with the corresponding
	basis vector.
\end{example}

Let \( V \) be a normed space.

\begin{definition}[Normalised vectors]
	A vector \( v \in V \) is called \emph{normalised} if
	\begin{equation}
		\|v\| = 1.
		\label{eq:normalised-vectors-1}
	\end{equation}
\end{definition}

Suppose \( v \in V \) is not normalised. But
\begin{equation}
	\frac{v}{\|v\|}
	\label{eq:orthogonality-6}
\end{equation}

is normalised. Because

\begin{equation}
	\left\|\frac{v}{\|v\|}\right\| =\footnote{Using absolute homogeneity property of norm} \frac{1}{\|v\|} \|v\| = 1.
	\label{eq:orthogonality-7}
\end{equation}

The process making a vector normalised in this way, is called \emph{normalisation}.

\begin{definition}[Orthogonal basis]
	A set of basis \( \{v_1, v_2, \ldots, v_n\} \) of a vector space \( V \) is called \emph{orthogonal basis} if
	\begin{equation}
		\langle v_i, v_j \rangle = \begin{cases}
			c \in \mathbb{R} & \text{if } i = j     \\
			0                & \text{if } i \neq j.
		\end{cases}
		\label{eq:orthogonal-basis-1}
	\end{equation}
\end{definition}

\begin{definition}[Orthonormal basis]
	A set of basis \( \{v_1, v_2, \ldots, v_n\} \) of a vector space \( V \) is called \emph{orthonormal basis} if
	\begin{equation}
		\langle v_i, v_j \rangle = \begin{cases}
			1 & \text{if } i = j     \\
			0 & \text{if } i \neq j.
		\end{cases}
		\label{eq:orthogonal-basis-2}
	\end{equation}
\end{definition}

\textbf{Notation (Kronecker/Dirac delta).}
The Kronecker delta is defined by
\begin{equation}
	\delta_{ij}=
	\begin{cases}
		1, & i=j,     \\
		0, & i\neq j.
	\end{cases}
	\label{eq:orthogonality-8}
\end{equation}
For example,
\[
	\delta_{00}=1,\qquad
	\delta_{11}=1,\qquad
	\delta_{10}=0.
\]
Using this notation, the defining condition for an
orthonormal basis \(\{v_1,\ldots,v_n\}\) becomes
\[
	\langle v_i,v_j\rangle=\delta_{ij}.
\]

In a sum, terms with unequal indices in the
Kronecker delta vanish. The remaining terms have
equal indices, so one summation can be removed.

\begin{example}
	Let
	\[
		\mathbf{x}=
		\begin{bmatrix}1\\-2\end{bmatrix}.
	\]
	Consider
	\[
		\sum_{i=1}^{2}\sum_{j=1}^{2}
		x_i\delta_{ij}x_j.
	\]
	Since \(\delta_{ij}=0\) unless \(i=j\),
	\[
		\sum_{i=1}^{2}\sum_{j=1}^{2}
		x_i\delta_{ij}x_j
		=\sum_{i=1}^{2}x_i\cdot1\cdot x_i.
	\]
	Expanding the original sum explicitly,
	\begin{align*}
		\sum_{i=1}^{2}\sum_{j=1}^{2}
		x_i\delta_{ij}x_j
		 & =
		x_1\underbrace{\delta_{11}}_{1}x_1
		+x_1\underbrace{\delta_{12}}_{0}x_2 \\
		 & \quad
		+x_2\underbrace{\delta_{21}}_{0}x_1
		+x_2\underbrace{\delta_{22}}_{1}x_2 \\
		 & =x_1^2+x_2^2                     \\
		 & =1^2+(-2)^2=5.
	\end{align*}
\end{example}

\begin{example}
	Let
	\[
		A=\begin{bmatrix}1&2\\3&4\end{bmatrix},
		\qquad
		\mathbf{x}=\begin{bmatrix}1\\-2\end{bmatrix}.
	\]
	Here, \(A_{ij}\) denotes the entry in row \(i\)
	and column \(j\). For example,
	\(A_{11}=1\) and \(A_{21}=3\).

	Consider
	\[
		\sum_{i=1}^{2}\sum_{j=1}^{2}\sum_{k=1}^{2}
		A_{ij}\delta_{jk}x_k.
	\]
	Only terms with \(j=k\) remain. Replacing \(j\)
	by \(k\), we obtain
	\begin{align*}
		\sum_{i=1}^{2}\sum_{j=1}^{2}\sum_{k=1}^{2}
		A_{ij}\delta_{jk}x_k
		 & =
		\sum_{i=1}^{2}\sum_{k=1}^{2}A_{ik}x_k \\
		 & =A_{11}x_1+A_{12}x_2               \\
		 & \quad+A_{21}x_1+A_{22}x_2          \\
		 & =1\cdot1+2\cdot(-2)                \\
		 & \quad+3\cdot1+4\cdot(-2)           \\
		 & =1-4+3-8                           \\
		 & =-8.
	\end{align*}
\end{example}

\begin{example}
	For the same matrix
	\[
		A=\begin{bmatrix}1&2\\3&4\end{bmatrix},
	\]
	consider
	\[
		\sum_{i=1}^{2}\sum_{j=1}^{2}\sum_{k=1}^{2}
		A_{ij}\delta_{ik}A_{jk}.
	\]
	Only terms with \(i=k\) remain. Replacing \(k\)
	by \(i\), we obtain
	\begin{align*}
		\sum_{i=1}^{2}\sum_{j=1}^{2}\sum_{k=1}^{2}
		A_{ij}\delta_{ik}A_{jk}
		 & =
		\sum_{i=1}^{2}\sum_{j=1}^{2}A_{ij}A_{ji} \\
		 & =A_{11}A_{11}+A_{12}A_{21}            \\
		 & \quad+A_{21}A_{12}+A_{22}A_{22}       \\
		 & =1^2+2\cdot3+3\cdot2+4^2              \\
		 & =1+6+6+16                             \\
		 & =29.
	\end{align*}
\end{example}

In addition,

\begin{theorem}
	Eigenvectors with different eigenvalues of a symmetric matrix \( A \) are orthogonal.
\end{theorem}
\begin{proof}
	Let \(A\) be a real symmetric matrix, and let
	\(\mathbf{x},\mathbf{y}\) be eigenvectors with
	distinct eigenvalues \(\lambda_1,\lambda_2\).
	Thus,
	\[
		A\mathbf{x}=\lambda_1\mathbf{x},
		\qquad
		A\mathbf{y}=\lambda_2\mathbf{y},
		\qquad
		\lambda_1\neq\lambda_2.
	\]

	On the one hand,
	\begin{align*}
		\langle A\mathbf{x},\mathbf{y}\rangle
		 & =(\lambda_1\mathbf{x})^T\mathbf{y}             \\
		 & =\lambda_1\mathbf{x}^T\mathbf{y}               \\
		 & =\lambda_1\langle\mathbf{x},\mathbf{y}\rangle.
	\end{align*}
	On the other hand, since \(A^T=A\),
	\begin{align*}
		\langle A\mathbf{x},\mathbf{y}\rangle
		 & =(A\mathbf{x})^T\mathbf{y}                     \\
		 & =\mathbf{x}^TA^T\mathbf{y}                     \\
		 & =\mathbf{x}^TA\mathbf{y}                       \\
		 & =\mathbf{x}^T(\lambda_2\mathbf{y})             \\
		 & =\lambda_2\langle\mathbf{x},\mathbf{y}\rangle.
	\end{align*}
	Therefore,
	\[
		(\lambda_1-\lambda_2)
		\langle\mathbf{x},\mathbf{y}\rangle=0.
	\]
	Since \(\lambda_1\neq\lambda_2\), we conclude that
	\[
		\langle\mathbf{x},\mathbf{y}\rangle=0.
	\]
	Hence, \(\mathbf{x}\) and \(\mathbf{y}\)
	are orthogonal.
\end{proof}

\begin{example}[Construction of a symmetric matrix]
	To construct a real symmetric matrix, choose
	its diagonal entries and the entries above
	the diagonal freely. Then reflect the entries
	above the diagonal across it, so that
	\(a_{ij}=a_{ji}\).

	For example, choosing the diagonal entries
	\(5,4,-3\) and the upper entries \(7,-2,13\)
	gives
	\[
		S=
		\begin{bmatrix}
			5  & 7  & -2 \\
			7  & 4  & 13 \\
			-2 & 13 & -3
		\end{bmatrix}.
	\]
	Taking the transpose,
	\[
		S^T=
		\begin{bmatrix}
			5  & 7  & -2 \\
			7  & 4  & 13 \\
			-2 & 13 & -3
		\end{bmatrix}
		=S.
	\]
	Hence, \(S\) is symmetric.
\end{example}

\begin{definition}[Anti/skew symmetric matrix]
	A matrix \( A \) is called  \emph{anti/skew symmetric} if
	\begin{equation}
		A = -A^T.
		\label{eq:anti-skew-symmetric-matrix-1}
	\end{equation}
\end{definition}

\begin{example}[Construction of a skew-symmetric matrix]
	For a real skew-symmetric matrix,
	\(a_{ij}=-a_{ji}\). In particular, every
	diagonal entry satisfies
	\[
		a_{ii}=-a_{ii}
		\implies a_{ii}=0.
	\]
	Thus, set the diagonal entries to zero,
	choose the entries above the diagonal freely,
	and place their negatives below the diagonal.

	For example,
	\[
		A=
		\begin{bmatrix}
			0  & 7  & -23 \\
			-7 & 0  & -13 \\
			23 & 13 & 0
		\end{bmatrix}.
	\]
	Taking the transpose,
	\[
		A^T=
		\begin{bmatrix}
			0   & -7  & 23 \\
			7   & 0   & 13 \\
			-23 & -13 & 0
		\end{bmatrix}
		=-A.
	\]
	Hence, \(A\) is skew-symmetric.
\end{example}

Most matrices are neither symmetric nor anti symmetric. But given a matrix \( M \), it can be written as sum of one symmetric and one anti symmetric matrix as follows:

\begin{equation}
	M = S_m + A_M,
	\label{eq:symmetric-skew-symmetric-matrix-1}
\end{equation}

where

\begin{equation}
	S_M = \frac{M + M^T}{2}\qquad\text{and}\qquad A_M = \frac{M - M^T}{2}.
	\label{eq:symmetric-skew-symmetric-matrix-2}
\end{equation}

\section{Review on multivariable calculus}

In this course, we will optimise functions whose inputs and outputs may consist of several real numbers. A function with several input variables is called a
\emph{multivariable function}. The branch of mathematics concerned with such functions is known as \emph{multivariable calculus}.

\subsection{Terminology Related to Functions}

\begin{definition}
	Let $A$ and $B$ be nonempty sets. A function $f:A\to B$ is a rule that assigns to every element $x\in A$ exactly one element $f(x)\in B$. Here,

	\begin{itemize}
		\item \( A \) is called the \emph{domain} of $f$, denoted by \( \mathrm{Dom} \ f \).

		\item \( B \) is called the \emph{codomain} of $f$.

		\item The \emph{range}, or \emph{image}, of $f$ is the subset of $B$ consisting of all values actually attained by $f$, namely the set $f(A)=\{f(x):x\in A\}$.

		\item The \emph{graph} of $f$ is the set $\operatorname{Graph}(f) =\{(x,f(x)):x\in A\}$.
	\end{itemize}
\end{definition}

\begin{example}
	Consider the function $f:\mathbb{R}\to\mathbb{R}$ defined by $f(x)=x^2$.

	Its domain and codomain are both $\mathbb{R}$, but range is \( [0, \infty) \).

\subsection{Directional Derivatives}

We first recall the notion of differentiation from elementary calculus.

\begin{definition}
	Let $I\subseteq\mathbb{R}$ be an open interval, let $f:I\to\mathbb{R}$, and	let $a\in I$. We say that $f$ is \emph{differentiable at $a$} if the following limit exists as a finite real number:
	\begin{equation}
		f'(a)
		=
		\lim_{h\to 0}
		\frac{f(a+h)-f(a)}{h}.
		\label{eq:single-variable-derivative}
	\end{equation}
	The value $f'(a)$ is called the \emph{derivative} of $f$ at $a$.
\end{definition}

Geometrically, the quotient $\frac{f(a+h)-f(a)}{h}$ represents the slope of the secant line joining $(a,f(a))$ and $(a+h,f(a+h))$. As $h$ approaches zero, the secant slope approaches the slope of the tangent line, provided the limit exists.

Suppose now that $f:U\to\mathbb{R}$, where $U$ is an open subset of $\mathbb{R}^n$ and $n\geq 2$.

Given $a\in U$, a small change in the input is represented by a vector $h\in\mathbb{R}^n$, rather than by a real number. One might attempt to imitate \eqref{eq:single-variable-derivative} by considering $\frac{f(a+h)-f(a)}{h}$.

However, this expression is not defined in the usual arithmetic of Euclidean spaces: its numerator is a real number, whereas its denominator is a vector. There is no canonical operation of division by a nonzero vector analogous to division by a
nonzero real number.

There is also a geometric difficulty. In $\mathbb{R}$, a point can be
approached only from the left or from the right. In $\mathbb{R}^n$, with $n\geq 2$, a point may be approached along infinitely many different directions. The rate of change of a function may depend on the particular direction in which its input is varied.

This leads us to the following definition:

\begin{definition}[Directional derivative]
	Let $U\subseteq\mathbb{R}^n$ be open, let
	$f:U\to\mathbb{R}^m$, and let $a\in U$.
	For a nonzero vector $v\in\mathbb{R}^n$, the
	\emph{directional derivative} of $f$ at $a$ in the direction 	$v$ is defined by
	\begin{equation}
		D_vf(a)
		=
		\lim_{t\to 0}
		\frac{f(a+tv)-f(a)}{t},
		\label{eq:directional-derivative}
	\end{equation}
	provided that this limit exists in $\mathbb{R}^m$.
\end{definition}

The directional derivative $D_vf(a)$ measures the instantaneous rate of change
of $f$ at $a$ as the input moves along the line $a+tv$. When $v$ is a unit vector, it measures the rate of change per unit distance in that direction.

\begin{remark}
	The ordinary derivative is a special case of the directional derivative. Indeed, when $n=m=1$ and $v=1$, the definition reduces to $D_1f(a)=f'(a)$.
\end{remark}
